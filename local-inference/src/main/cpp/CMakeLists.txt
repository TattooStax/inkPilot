cmake_minimum_required(VERSION 3.22.1)
project(inkpilot_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
foreach(feature AVX AVX2 AVX512 FMA F16C BMI2)
    set(GGML_${feature} OFF CACHE BOOL "Portable x86 emulator baseline" FORCE)
endforeach()

include(FetchContent)
FetchContent_Declare(llama_cpp
    URL https://github.com/ggml-org/llama.cpp/archive/7071ba378afa96dcd69c41e1ec2b4912364dbee4.tar.gz
    URL_HASH SHA256=93d453d0ad99853b9f1dafeea78fa55da9a838ba71804abcf1aa823537703b79)
FetchContent_MakeAvailable(llama_cpp)

add_library(inkpilot-engine STATIC inference_engine.cpp inference_generation.cpp inference_sampling.cpp)
target_link_libraries(inkpilot-engine PUBLIC llama-common)
target_include_directories(inkpilot-engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

if(ANDROID)
    add_library(inkpilot-inference SHARED inference_jni.cpp)
    target_link_libraries(inkpilot-inference PRIVATE inkpilot-engine)
    target_link_options(inkpilot-inference PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(inkpilot-model-qualification model_qualification.cpp)
    target_link_libraries(inkpilot-model-qualification PRIVATE inkpilot-engine)
    add_executable(inkpilot-tool-probe tool_protocol_probe.cpp)
    target_link_libraries(inkpilot-tool-probe PRIVATE inkpilot-engine)
    add_executable(inkpilot-inference-smoke inference_smoke.cpp)
    target_link_libraries(inkpilot-inference-smoke PRIVATE inkpilot-engine)
endif()
